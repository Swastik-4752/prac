# ===== Practical 7 - Flask =====

# ----- app.py -----
from flask import Flask, request, jsonify

app = Flask("Flask_Practical")

@app.route("/")
def home():
    return "Student Prediction API is Running"

@app.route("/predict", methods=["POST"])
def predict():
    data = request.get_json()
    if data["attendance"] >= 75 and data["study_hours"] >= 2 and data["previous_marks"] >= 50:
        data["prediction"] = "PASS"
    else:
        data["prediction"] = "NEEDS IMPROVEMENT"
    return jsonify(data)

app.run(debug=True)
