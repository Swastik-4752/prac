# prac - practical code loader for Windows (PowerShell 5.1+ and PowerShell 7)
# Run:   irm https://swastik-4752.github.io/prac/w.txt | iex
& {
    $ErrorActionPreference = 'Stop'
    $Base = 'https://swastik-4752.github.io/prac'
    if ($env:PRAC_BASE) { $Base = $env:PRAC_BASE.TrimEnd('/') }

    # Number given up front (prac 2) skips the menu and the questions.
    $pick = "$env:PRAC_N".Trim()
    Remove-Item Env:PRAC_N -ErrorAction SilentlyContinue
    $quick = [bool]$pick

    # Old Windows PowerShell needs TLS 1.2 switched on for https sites.
    try { [Net.ServicePointManager]::SecurityProtocol = [Net.ServicePointManager]::SecurityProtocol -bor 3072 } catch {}

    function Get-Text([string]$name) {
        $r = Invoke-WebRequest -UseBasicParsing -Uri "$Base/$name"
        if ($r.Content -is [byte[]]) { return [Text.Encoding]::UTF8.GetString($r.Content) }
        return [string]$r.Content
    }

    try { $raw = Get-Text 'list.txt' }
    catch { Write-Host "Could not reach $Base  (check your internet)" -ForegroundColor Red; return }

    $items = @()
    foreach ($line in ($raw -split "`r?`n")) {
        $t = $line.Trim()
        if (-not $t -or $t.StartsWith('#')) { continue }
        $p = $t.Split('|')
        if ($p.Count -lt 3) { continue }
        $items += [pscustomobject]@{ No = $p[0].Trim(); Title = $p[1].Trim(); File = $p[2].Trim() }
    }

    if (-not $quick) {
        Write-Host ''
        Write-Host '  ==== PRACTICALS ====' -ForegroundColor Cyan
        foreach ($i in $items) {
            Write-Host ('  {0,3}.  ' -f $i.No) -ForegroundColor Yellow -NoNewline
            Write-Host $i.Title
        }
        Write-Host ''
        $pick = (Read-Host '  Enter practical number (q to quit)').Trim()
        if (-not $pick -or $pick -eq 'q') { return }
    }

    $item = $items | Where-Object { $_.No -eq $pick } | Select-Object -First 1
    if (-not $item) { Write-Host "  No practical number '$pick'." -ForegroundColor Red; return }

    try { $code = Get-Text "$($item.No).txt" }
    catch { Write-Host "  Practical $($item.No) file is missing on the site." -ForegroundColor Red; return }
    $code = $code -replace "`r`n", "`n"

    Write-Host ''
    Write-Host "  ---- $($item.No). $($item.Title)  [$($item.File)] ----" -ForegroundColor Cyan
    Write-Host ''
    Write-Host $code
    Write-Host "  ---- end of $($item.File) ----" -ForegroundColor Cyan

    try { Set-Clipboard -Value $code; Write-Host '  Copied to clipboard.' -ForegroundColor Green } catch {}

    $dest = Join-Path (Get-Location).ProviderPath $item.File
    $save = $true
    if ($quick) {
        if (Test-Path -LiteralPath $dest) { $save = $false; Write-Host "  $($item.File) already exists here - not overwritten." -ForegroundColor Yellow }
    } else {
        $q = if (Test-Path -LiteralPath $dest) { "  $($item.File) already exists. Overwrite? (y/N)" } else { "  Save as $($item.File) in this folder? (Y/n)" }
        $a = (Read-Host $q).Trim().ToLower()
        if (Test-Path -LiteralPath $dest) { $save = ($a -eq 'y') } else { $save = ($a -ne 'n') }
    }
    if ($save) {
        # UTF-8 without BOM: javac rejects a BOM.
        [IO.File]::WriteAllText($dest, $code, (New-Object Text.UTF8Encoding $false))
        Write-Host "  Saved: $dest" -ForegroundColor Green
    }

    # ---- Run it: create all files + run all commands in a NEW Git Bash window, then close this one ----
    if ($quick) { return }
    Write-Host ''
    Write-Host '  Run it now? This creates all the files in this folder, runs every command' -ForegroundColor Yellow
    Write-Host '  (installs too) in a NEW terminal window, and closes this window.' -ForegroundColor Yellow
    if ((Read-Host '  Run practical now? (y/N)').Trim().ToLower() -ne 'y') { return }

    # The commands are Git Bash commands, so find Git Bash (comes with Git for Windows).
    $bash = $null
    $git = Get-Command git.exe -ErrorAction SilentlyContinue | Select-Object -First 1
    if ($git) { $bash = Join-Path (Split-Path (Split-Path $git.Source)) 'bin\bash.exe' }
    if (-not ($bash -and (Test-Path $bash))) { $bash = Join-Path $env:ProgramFiles 'Git\bin\bash.exe' }
    if (-not (Test-Path $bash)) {
        Write-Host '  Git Bash not found. Install Git for Windows (git-scm.com), then try again.' -ForegroundColor Red
        return
    }

    $tmp = Join-Path $env:TEMP 'prac'
    New-Item -ItemType Directory -Force -Path $tmp | Out-Null
    $codeFile = Join-Path $tmp "practical$($item.No).txt"
    $runFile = Join-Path $tmp 'run.sh'
    $utf8 = New-Object Text.UTF8Encoding $false
    [IO.File]::WriteAllText($codeFile, $code, $utf8)
    try { [IO.File]::WriteAllText($runFile, ((Get-Text 'run.txt') -replace "`r`n", "`n"), $utf8) }
    catch { Write-Host '  Could not download the runner.' -ForegroundColor Red; return }

    $here = (Get-Location).ProviderPath
    $q = { param($p) '"' + ($p -replace '\\', '/') + '"' }
    $env:CHERE_INVOKING = '1'   # keep Git Bash in this folder instead of jumping to home
    Start-Process -FilePath $bash -WorkingDirectory $here -ArgumentList ('--login ' + (& $q $runFile) + ' ' + (& $q $codeFile) + ' ' + (& $q $here))
    Write-Host '  Started in a new window. Closing this one...' -ForegroundColor Green
    Start-Sleep -Seconds 2

    # Close this terminal: stop this shell and the shells that started it (cmd / PowerShell).
    $ids = @($PID)
    $p = Get-CimInstance Win32_Process -Filter "ProcessId=$PID"
    while ($p) {
        $parent = Get-CimInstance Win32_Process -Filter "ProcessId=$($p.ParentProcessId)"
        if (-not $parent -or $parent.Name -notmatch '^(powershell|pwsh|cmd)\.exe$') { break }
        $ids += $parent.ProcessId
        $p = $parent
    }
    if ($env:PRAC_NOCLOSE) { Write-Host "  (test) would close: $($ids -join ', ')"; return }
    [array]::Reverse($ids)
    foreach ($id in $ids) { Stop-Process -Id $id -Force -ErrorAction SilentlyContinue }
}
