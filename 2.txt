git config --global user.name "Your Name"
git config --global user.email "you@example.com"
mkdir ml_project
cd ml_project
git init
git branch -M master
echo 'print("Initial ML Model")' > train.py
git add .
git commit -m "Initial commit"
git checkout -b feature-model
echo 'print("Feature Version")' > train.py
git commit -am "Feature update"
git checkout master
echo 'print("Master Version")' > train.py
git commit -am "Master update"
git merge feature-model
# conflict! keep both print lines in train.py:
printf 'print("Master Version")\nprint("Feature Version")\n' > train.py
git add train.py
git commit -m "Merge feature-model into master"
git log --oneline --graph --all
