# ===== Practical 3 - DVC (run in Git Bash) =====
pip install dvc
mkdir dvc-demo
cd dvc-demo
git init
dvc init
echo "This is my dataset" > data.txt
dvc add data.txt
git add data.txt.dvc .gitignore
git commit -m "Track data with DVC"
mkdir storage
dvc remote add -d localstore storage
dvc push
rm data.txt
dvc pull
cat data.txt
